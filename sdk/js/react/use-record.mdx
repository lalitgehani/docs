---
title: "useRecord Hook"
description: "Fetch a single record in React"
---

## Import

```ts
import { useRecord } from "@snackbase/react";
```

## Usage

```tsx
function PostDetail({ id }: { id: string }) {
  const { data, loading, error, refetch } = useRecord("posts", id);

  if (loading) return <p>Loading...</p>;
  if (error) return <p>{error.message}</p>;
  if (!data) return <p>Not found</p>;

  return (
    <article>
      <h1>{data.title}</h1>
      <button onClick={() => refetch()}>Reload</button>
    </article>
  );
}
```

## Options

```ts
useRecord(collection, id, {
  fields?: string[] | string;
  expand?: string[] | string;
  enabled?: boolean;              // default true when id is non-empty
  listenToInvalidation?: boolean; // default true
})
```

Empty `id` skips the network call. Stale in-flight responses after `id` changes are ignored.
